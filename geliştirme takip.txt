                              ENTITY 1: Page
================================================================================
Açıklama: Ana sayfa entity'si - Tüm sayfaların temel bilgilerini tutar

public class Page
{
    // Primary Key
    public int Id { get; set; }

    // Temel Bilgiler
    public string Name { get; set; }              // Sayfa adı (örn: "Ana Sayfa")
    public string Slug { get; set; }              // URL slug (örn: "about", "blog")
    public string Url { get; set; }               // Tam URL
    public PageType PageType { get; set; }        // Sayfa tipi (enum)

    // SEO Ayarları
    public double Priority { get; set; }          // Sitemap priority (0.0 - 1.0)
    public bool IsInSitemap { get; set; }         // Sitemap'te var mı?
    public bool IsDynamic { get; set; }           // Dinamik sayfa mı?
    public bool IsActive { get; set; }            // Aktif mi?

    // Zaman Damgaları
    public DateTime CreatedAt { get; set; }
    public DateTime UpdatedAt { get; set; }

    // Navigation Properties
    public MetaTag MetaTag { get; set; }
    public OpenGraph OpenGraph { get; set; }
    public TwitterCard TwitterCard { get; set; }
    public StructuredData StructuredData { get; set; }
    public SitemapEntry SitemapEntry { get; set; }
}

// Sayfa Tipi Enum
public enum PageType
{
    Home = 0,           // Ana Sayfa
    About = 1,          // Hakkımda
    Blog = 2,           // Blog Listesi
    BlogDetail = 3,     // Blog Detay (Dinamik)
    Work = 4,           // Projeler Listesi
    WorkDetail = 5,     // Proje Detay (Dinamik)
    Skills = 6,         // Yetenekler
    Music = 7,          // Müzik (Dinamik)
    Contact = 8,        // İletişim
    Custom = 99         // Özel Sayfa
}

================================================================================
                              ENTITY 2: MetaTag
================================================================================
Açıklama: Sayfa meta tag bilgilerini tutar

public class MetaTag
{
    // Primary Key
    public int Id { get; set; }

    // Foreign Key
    public int PageId { get; set; }

    // Meta Tag Bilgileri
    public string Title { get; set; }             // <title> tag
    public string Description { get; set; }       // meta description (max 160)
    public string Keywords { get; set; }          // meta keywords (virgülle ayrılmış)
    public string Author { get; set; }            // meta author
    public string Robots { get; set; }            // meta robots (index, follow vb.)
    public string CanonicalUrl { get; set; }      // canonical URL
    public string Viewport { get; set; }          // viewport (varsayılan değer)
    public string Charset { get; set; }           // charset (varsayılan: UTF-8)
    public string Language { get; set; }          // html lang attribute

    // Ek Meta Taglar (JSON olarak)
    public string AdditionalTags { get; set; }    // Ek özel meta taglar (JSON)

    // Navigation Property
    public Page Page { get; set; }
}

================================================================================
                              ENTITY 3: OpenGraph
================================================================================
Açıklama: Open Graph (Facebook/LinkedIn) meta bilgilerini tutar

public class OpenGraph
{
    // Primary Key
    public int Id { get; set; }

    // Foreign Key
    public int PageId { get; set; }

    // Open Graph Bilgileri
    public string OgType { get; set; }            // website, article, profile, music.song
    public string OgUrl { get; set; }             // og:url
    public string OgTitle { get; set; }           // og:title
    public string OgDescription { get; set; }     // og:description
    public string OgImage { get; set; }           // og:image (1200x630 px)
    public string OgImageAlt { get; set; }        // og:image:alt
    public int? OgImageWidth { get; set; }        // og:image:width
    public int? OgImageHeight { get; set; }       // og:image:height
    public string OgLocale { get; set; }          // og:locale (tr_TR)
    public string OgSiteName { get; set; }        // og:site_name

    // Article Spesifik (Blog için)
    public string ArticleAuthor { get; set; }     // article:author
    public DateTime? ArticlePublishedTime { get; set; }  // article:published_time
    public DateTime? ArticleModifiedTime { get; set; }   // article:modified_time
    public string ArticleSection { get; set; }    // article:section
    public string ArticleTags { get; set; }       // article:tag (virgülle ayrılmış)

    // Music Spesifik (Müzik için)
    public string MusicDuration { get; set; }     // music:duration
    public string MusicAlbum { get; set; }        // music:album
    public string MusicMusician { get; set; }     // music:musician

    // Navigation Property
    public Page Page { get; set; }
}

public class StructuredData
{
    // Primary Key
    public int Id { get; set; }

    // Foreign Key
    public int PageId { get; set; }

    // Schema Bilgileri
    public SchemaType SchemaType { get; set; }    // Schema tipi (enum)
    public string JsonLdContent { get; set; }     // JSON-LD içeriği (tam JSON)
    public bool IsActive { get; set; }            // Aktif mi?

    // Zaman Damgaları
    public DateTime CreatedAt { get; set; }
    public DateTime UpdatedAt { get; set; }

    // Navigation Property
    public Page Page { get; set; }
}

// Schema Tipi Enum
public enum SchemaType
{
    Person = 0,              // Kişi
    Organization = 1,        // Organizasyon
    WebSite = 2,             // Web Sitesi
    WebPage = 3,             // Web Sayfası

public class ApplicationDbContext : DbContext, IApplicationDbContext
{
    public ApplicationDbContext(DbContextOptions<ApplicationDbContext> options)
        : base(options) { }

    // DbSet'ler
    public DbSet<Page> Pages { get; set; }
    public DbSet<MetaTag> MetaTags { get; set; }
    public DbSet<OpenGraph> OpenGraphs { get; set; }
    public DbSet<TwitterCard> TwitterCards { get; set; }
    public DbSet<StructuredData> StructuredDatas { get; set; }
    public DbSet<SitemapEntry> SitemapEntries { get; set; }
    public DbSet<SocialMedia> SocialMedias { get; set; }
    public DbSet<SiteSettings> SiteSettings { get; set; }
    public DbSet<Blog> Blogs { get; set; }
    public DbSet<Project> Projects { get; set; }
    public DbSet<Music> Musics { get; set; }
    public DbSet<AuditLog> AuditLogs { get; set; }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.ApplyConfigurationsFromAssembly(Assembly.GetExecutingAssembly());
        base.OnModelCreating(modelBuilder);
    }

    public override Task<int> SaveChangesAsync(CancellationToken cancellationToken = default)
    {
        // Audit logging
        foreach (var entry in ChangeTracker.Entries<BaseEntity>())
        {
            switch (entry.State)
            {
                case EntityState.Added:
                    entry.Entity.CreatedAt = DateTime.UtcNow;
                    entry.Entity.UpdatedAt = DateTime.UtcNow;
                    break;
                case EntityState.Modified:
                    entry.Entity.UpdatedAt = DateTime.UtcNow;
                    break;
            }
        }
        return base.SaveChangesAsync(cancellationToken);
    }
}

[ApiController]
[Route("api/[controller]")]
public class PagesController : ControllerBase
{
    private readonly IMediator _mediator;

    public PagesController(IMediator mediator)
    {
        _mediator = mediator;
    }

    [HttpGet]
    public async Task<ActionResult<List<PageDto>>> GetAll()
    {
        var result = await _mediator.Send(new GetAllPagesQuery());
        return Ok(result);
    }

    [HttpGet("{id}")]
    public async Task<ActionResult<PageDto>> GetById(int id)
    {
        var result = await _mediator.Send(new GetPageByIdQuery(id));
        return result != null ? Ok(result) : NotFound();
    }

    [HttpGet("slug/{slug}")]
    public async Task<ActionResult<PageDto>> GetBySlug(string slug)
    {
        var result = await _mediator.Send(new GetPageBySlugQuery(slug));
        return result != null ? Ok(result) : NotFound();
    }

    [HttpPost]
    public async Task<ActionResult<int>> Create(CreatePageCommand command)
    {
        var result = await _mediator.Send(command);
        return CreatedAtAction(nameof(GetById), new { id = result }, result);
    }

    [HttpPut("{id}")]
    public async Task<IActionResult> Update(int id, UpdatePageCommand command)
    {
        if (id != command.Id) return BadRequest();
        await _mediator.Send(command);
        return NoContent();
    }

    [HttpDelete("{id}")]
    public async Task<IActionResult> Delete(int id)
    {
        await _mediator.Send(new DeletePageCommand(id));
        return NoContent();
    }
}

{

    public int Id { get; set; }

    // Temel Bilgiler
    public string Name { get; set; }              // "Mehmet Asker"

    public string LogoUrl { get; set; }           // Site logosu
    public string FaviconUrl { get; set; }        // Favicon

    // Durum
    public bool IsActive { get; set; }            // Aktif mi?
    public bool IsDefault { get; set; }           // Varsayılan site mi?
    public DateTime CreatedAt { get; set; }
    public DateTime UpdatedAt { get; set; }

    // İlişkiler - Her sitenin kendi verileri
    public SiteSettings Settings { get; set; }
    public ICollection<Page> Pages { get; set; }
    public ICollection<Blog> Blogs { get; set; }
    public ICollection<Project> Projects { get; set; }
    public ICollection<Music> Musics { get; set; }
    public ICollection<SocialMedia> SocialMedias { get; set; }
}

================================================================================
                         FRONTEND STRATEJİSİ
================================================================================

[SEÇENEK 1: REACT SPA (Mevcut 2.sitem)]
---------------------------------------
- 2.sitem React uygulaması olduğu gibi kalır
- API'den veri çeker
- Build edilip static olarak sunulur
- Her site için ayrı build veya dinamik config

Avantajları:
  + Mevcut kod kullanılır
  + React ekosistemi
  + Zengin animasyonlar (Framer Motion)

Dezavantajları:
  - Her site için build gerekebilir
  - SEO için SSR/Pre-rendering gerekli

[SEÇENEK 2: NEXT.JS (SSR + SSG)]
--------------------------------
- Server-side rendering
- Static site generation
- API routes
- SEO dostu

Avantajları:
  + SEO mükemmel
  + Performans yüksek
  + Hibrit rendering

Dezavantajları:
  - Mevcut kod migration gerekir
  - Öğrenme eğrisi

[SEÇENEK 3: STATIK SITE + API]
------------------------------
- HTML/CSS/JS statik dosyalar
- JavaScript ile API'den veri çekme
- Her site için ayrı template

Avantajları:
  + En basit
  + Hızlı
  + CDN ile kolay dağıtım

Dezavantajları:
  - Dinamik özellikler sınırlı
  - Her değişiklikte rebuild

[ÖNERİLEN: SEÇENEK 1 + PRE-RENDERING]
-------------------------------------
- React (mevcut 2.sitem) kullan
- react-snap veya prerender.io ile pre-rendering
- API'den SEO verileri çek
- react-helmet ile dinamik meta taglar
